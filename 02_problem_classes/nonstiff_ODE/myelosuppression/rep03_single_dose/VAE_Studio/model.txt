[PARAMETERS]
Cl_C
V_C
Circ0
MTT
gamma
Slope

[PK]
; analytic PK models from the built-in library
C = pkmodel(1cmt_cl, dose = Dose)

[EQUATIONS]
ktr = 4/MTT
Edrug = Slope*C
ddt_Prol = ktr*Prol*(1 - Edrug)*(Circ0/Circ)^gamma - ktr*Prol
Prol(0) = Circ0
ddt_T1 = ktr*(Prol - T1)
T1(0) = Circ0
ddt_T2 = ktr*(T1 - T2)
T2(0) = Circ0
ddt_T3 = ktr*(T2 - T3)
T3(0) = Circ0
ddt_Circ = ktr*(T3 - Circ)
Circ(0) = Circ0

[OUTPUT]
; <observation in the dataset> = <model variable>
Conc = C
ANC = Circ
