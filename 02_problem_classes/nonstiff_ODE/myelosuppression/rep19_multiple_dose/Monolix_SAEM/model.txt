DESCRIPTION: Myelosuppression model (Friberg et al. 2002), 1-cmt IV PK (closed form) + 5 transit states

[LONGITUDINAL]
input = {V, Cl, Circ0, MTT, gamma, Slope}

PK:
Cc = pkmodel(V, Cl)

EQUATION:
t_0 = 0
Prol_0 = Circ0
T1_0 = Circ0
T2_0 = Circ0
T3_0 = Circ0
Circ_0 = Circ0
ktr = 4/MTT
Edrug = Slope*Cc
ddt_Prol = ktr*Prol*(1 - Edrug)*(Circ0/Circ)^gamma - ktr*Prol
ddt_T1 = ktr*(Prol - T1)
ddt_T2 = ktr*(T1 - T2)
ddt_T3 = ktr*(T2 - T3)
ddt_Circ = ktr*(T3 - Circ)

OUTPUT:
output = {Cc, Circ}
