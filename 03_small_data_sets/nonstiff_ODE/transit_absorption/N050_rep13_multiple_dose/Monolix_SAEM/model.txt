DESCRIPTION: Transit-compartment absorption (Savic et al. 2007), n = 3 transit compartments, 1-cmt

[LONGITUDINAL]
input = {MTT, ka, V, Cl}

PK:
depot(target = T1)

EQUATION:
t_0 = 0
T1_0 = 0
T2_0 = 0
T3_0 = 0
Ad_0 = 0
Ac_0 = 0
ktr = 4/MTT
ddt_T1 = -ktr*T1
ddt_T2 = ktr*(T1 - T2)
ddt_T3 = ktr*(T2 - T3)
ddt_Ad = ktr*T3 - ka*Ad
ddt_Ac = ka*Ad - Cl/V*Ac
Cc = Ac/V

OUTPUT:
output = {Cc}
