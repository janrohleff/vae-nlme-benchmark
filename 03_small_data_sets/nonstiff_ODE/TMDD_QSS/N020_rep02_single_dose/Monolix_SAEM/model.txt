DESCRIPTION: TMDD, quasi-steady-state approximation, 2-cmt, IV bolus (time d, amount nmol, conc nM)

[LONGITUDINAL]
input = {Cl, V1, Q, V2, R0, kdeg, kint, KSS}

PK:
depot(target = Atot)

EQUATION:
t_0 = 0
Atot_0 = 0
A2_0 = 0
Rtot_0 = R0
Ctot = Atot/V1
Dq = Ctot - Rtot - KSS
C = 0.5*(Dq + sqrt(Dq^2 + 4*KSS*Ctot))
Occ = Rtot*C/(KSS + C)
ddt_Atot = -Cl*C - Q*C + Q*A2/V2 - kint*V1*Occ
ddt_A2 = Q*C - Q*A2/V2
ddt_Rtot = kdeg*R0 - kdeg*Rtot - (kint - kdeg)*Occ

OUTPUT:
output = {C, Rtot}
