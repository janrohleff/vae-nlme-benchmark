DESCRIPTION: TMDD, full binding model (Mager & Jusko 2001), 2-cmt, IV bolus - stiff (time d, amount nmol, conc nM)

[LONGITUDINAL]
input = {Cl, V1, Q, V2, R0, kdeg, kint, KD}

PK:
depot(target = A1)

EQUATION:
odeType = stiff
odeRelTol = 1e-07
odeAbsTol = 1e-10
t_0 = 0
A1_0 = 0
A2_0 = 0
R_0 = R0
P_0 = 0
kon = 10
koff = KD*kon
C = A1/V1
Bind = kon*C*R - koff*P
ddt_A1 = -Cl*C - Q*C + Q*A2/V2 - Bind*V1
ddt_A2 = Q*C - Q*A2/V2
ddt_R = kdeg*R0 - kdeg*R - Bind
ddt_P = Bind - kint*P
Rtot = R + P

OUTPUT:
output = {C, Rtot}
