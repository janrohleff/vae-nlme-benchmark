DESCRIPTION: 1-cmt oral PK (closed form) + receptor binding kinetics + indirect response (stiff)

[LONGITUDINAL]
input = {ka, V, Cl, R0, kout, KD}

PK:
Cc = pkmodel(ka, V, Cl)

EQUATION:
odeType = stiff
odeRelTol = 0.0001
odeAbsTol = 1e-07
t_0 = 0
Occ_0 = 0
R_0 = R0
kon = 50
koff = KD*kon
kin = R0*kout
ddt_Occ = kon*Cc*(1 - Occ) - koff*Occ

OUTPUT:
output = {Cc, R}
